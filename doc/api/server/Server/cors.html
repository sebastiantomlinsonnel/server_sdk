Middleware that lets browsers on other origins call the API, and answers OPTIONS preflight requests.

origin is sent as access-control-allow-origin; the default * allows any site. methods and headers list what callers may use.

Server(
  name: 'api',
  middleware: [Server.cors(origin: 'https://example.com')],
  routes: [...],
);

Implementation

static Middleware cors({
  String origin = '*',
  List<String> methods = const ['GET', 'POST', 'PUT', 'PATCH', 'DELETE'],
  List<String> headers = const ['content-type', 'authorization'],
}) {
  final corsHeaders = {
    'access-control-allow-origin': origin,
    'access-control-allow-methods': [...methods, 'OPTIONS'].join(', '),
    'access-control-allow-headers': headers.join(', '),
  };

  return (inner) => (request) async {
    if (request.method == 'OPTIONS') {
      return Response.ok(null, headers: corsHeaders);
    }
    final response = await inner(request);

    return response.change(headers: corsHeaders);
  };
}
