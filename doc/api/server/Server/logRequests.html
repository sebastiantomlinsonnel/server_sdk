Middleware logRequests()

Middleware that logs every request with cmdline, like GET /api/todos?done=false 200 3ms.

The level follows the answer: log for success and redirects, warn for 4xx, and fail for 5xx and for errors thrown past the API (like a failing website).

Server(
  name: 'api',
  middleware: [Server.logRequests()],
  routes: [...],
);

Implementation

static Middleware logRequests() =>
    (inner) => (request) async {
      final watch = Stopwatch()..start();
      final query = request.url.hasQuery ? '?${request.url.query}' : '';
      final line = '${request.method} /${request.url.path}$query';
      String took() => '${watch.elapsedMilliseconds}ms';

      try {
        final response = await inner(request);
        final status = response.statusCode;
        final message = '$line $status ${took()}';
        if (status >= 500) {
          fail(message);
        } else if (status >= 400) {
          warn(message);
        } else {
          log(message);
        }

        return response;
      } on HijackException {
        rethrow;
      } catch (error) {
        fail('$line threw after ${took()}: $error');
        rethrow;
      }
    };
